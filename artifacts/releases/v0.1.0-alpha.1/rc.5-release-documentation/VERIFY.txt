FPR=0E6699CB48F3941AE28CA88D509F78007F62D5EA
curl --connect-timeout 10 --max-time 30 --fail --silent --show-error --proto '=https' --tlsv1.2   'https://openpgpkey.nachtigall.dev/.well-known/openpgpkey/nachtigall.dev/hu/rckmfa7rw857sf15hkixzhxx4681turz?l=christian'   --output "$VERIFY_HOME/publisher-key"
